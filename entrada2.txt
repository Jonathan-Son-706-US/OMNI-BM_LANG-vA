MAIN {
    DEC $a, $b, $c, $total, $i;
    
    #! Lectura y asignación básica !#
    READ $b;
    READ $c;
    $a => 1 + 2 * $b;
    
    $total => $c : 2;  
    IF ($total < 25 && $b > 0 || $a == 10) {
        IF ($total <> 5) {
            $total => $total + -1;
        } ELSE {
            $total => 0;
        }
    } ELSE {
        $total => 100;
    }
    WHILE ($total > 50 && $i < 5) {
        $total => $total - 5;
        $i => $i + 1;
    }
    FOR ($i => 0; $i < 10; $i => $i+1) {
        $total => $total + 2;
    }
    
    WRITE $total;
}
